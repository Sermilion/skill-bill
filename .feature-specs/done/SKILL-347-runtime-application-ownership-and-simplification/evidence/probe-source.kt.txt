class ApplicationInvestigationProbeTest {
  @Test
  fun `activity cache suppresses a second database after first write failed`() {
    val first = FailingDatabase(IOException("unavailable"))
    val second = FailingDatabase(IOException("different database"))
    val clock = Clock.fixed(Instant.parse("2026-09-15T12:00:00Z"), ZoneOffset.UTC)
    val workflow = UUID.randomUUID().toString()
    AgentActivityStampWriter(first, clock).recordEvidenceRead(workflow, null)
    AgentActivityStampWriter(second, clock).recordEvidenceRead(workflow, null)
    assertEquals(1, first.calls)
    assertEquals(0, second.calls)
  }

  @Test
  fun `activity write cancellation is swallowed`() {
    val database = FailingDatabase(CancellationException("cancelled"))
    AgentActivityStampWriter(database, Clock.systemUTC()).recordEvidenceRead(UUID.randomUUID().toString(), null)
    assertEquals(1, database.calls)
  }

  @Test
  fun `spec read cancellation becomes source unavailable`() {
    val fileStore = object : DecompositionManifestStore by TestDecompositionManifestStore {
      override fun isRegularFile(path: Path) = true
      override fun readText(path: Path): String = throw CancellationException("cancelled")
    }
    val validator = object : ReviewContextEnvelopeValidator {
      override fun validate(envelope: ReviewContextWireMap, sourceLabel: String) = Unit
      override fun validateSpecIntentProjection(envelope: ReviewContextWireMap, sourceLabel: String) = Unit
    }
    val error = assertFailsWith<SpecIntentSourceUnavailable> {
      SpecIntentProjectionExtractor(validator, fileStore).extract(
        Path.of("/tmp/audit-repo"), Path.of("spec.md"), ReviewContextBudgetPolicy.DEFAULT, explicit = false,
      )
    }
    assertEquals("unreadable", error.reason)
  }

  @Test
  fun `optional persistence turns interruption into fallback`() {
    val boundary = RuntimeOwnedPersistenceBoundary(FailingDatabase(InterruptedException("cancelled")), NoopRuntimeDiagnostics)
    val result = boundary.optionalRead("probe", "fact", "fallback") { "unreachable" }
    assertEquals("fallback", result)
  }

  @Test
  fun `update check turns interruption into unknown`() {
    val settings = object : TelemetrySettingsProvider {
      override fun load(materialize: Boolean): TelemetrySettings = error("unused")
    }
    val service = UpdateCheckService(
      SystemService(FailingDatabase(IOException("unused")), settings, "0.3.0"),
      TransportContext(requester = RemoteTransportPort { _, _, _, _ -> throw InterruptedException("cancelled") }),
    )
    val result = service.check(false)
    assertEquals(UpdateCheckStatus.UNKNOWN, result.status)
    assertTrue(result.reason.orEmpty().contains("network failure"))
  }

  @Test
  fun `cursor staging failure leaves bound endpoint unclosed`() {
    var opened = 0
    var closed = 0
    val delegate = stubGovernedReviewEvidenceEndpointBinder(Files.createTempDirectory("audit-endpoint"))
    val binder = object : GovernedReviewEvidenceEndpointBinder {
      override fun bind(lane: String, protocol: NativeReviewOperationProtocol, onEvidenceRead: (() -> Unit)?): GovernedReviewEvidenceEndpointHandle {
        opened++
        val handle = delegate.bind(lane, protocol, onEvidenceRead)
        return object : GovernedReviewEvidenceEndpointHandle by handle {
          override fun close() { closed++; handle.close() }
        }
      }
    }
    val recorder = ReviewRecorder()
    val runner = reviewHarness(
      ReviewHarnessConfig(
        manifests = listOf(reviewPack("kotlin", listOf("architecture"), routingSignals = listOf("*.kt"))),
        diff = diffForPaths("src/Repo.kt"),
        evidenceEndpointBinder = binder,
      ), recorder,
    )
    val laneField = runner.javaClass.getDeclaredField("laneLaunch").apply { isAccessible = true }
    val lane = laneField.get(runner)
    lane.javaClass.getDeclaredField("reviewLaunchAgentStaging").apply { isAccessible = true }
      .set(lane, ReviewLaunchAgentStagingPort { throw IOException("staging failed") })
    val result = runner.run(harnessRequest(agent1Id = "cursor", codeReviewMode = CodeReviewExecutionMode.DELEGATED))
    assertEquals(1, opened)
    assertEquals(0, closed)
    assertTrue(result.lane1.failureReason.orEmpty().contains("staging failed"))
    assertTrue(recorder.parentLaunches.isEmpty())
  }
}

private class FailingDatabase(private val failure: Exception) : DatabaseSessionFactory {
  var calls = 0
  override fun resolveDbPath(): Path = Path.of("/tmp/unused-audit.db")
  override fun databaseExists(): Boolean = true
  override fun <T> read(block: (UnitOfWork) -> T): T { calls++; throw failure }
  override fun <T> transaction(block: (UnitOfWork) -> T): T { calls++; throw failure }
  override fun <T> selfManagedWrite(block: (UnitOfWork) -> T): T { calls++; throw failure }
}
